cmake_minimum_required(VERSION 3.20)
project(LGEngineCppClient LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "LGEngine requires Windows.")
endif()
add_library(LGEngineCppClient INTERFACE)
add_library(LGEngine::Client ALIAS LGEngineCppClient)
target_include_directories(LGEngineCppClient INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(LGEngineCppClient INTERFACE cxx_std_17)
target_link_libraries(LGEngineCppClient INTERFACE user32 advapi32 bcrypt)
option(LGENGINE_BUILD_EXAMPLE "Build the standalone C++ consumer" OFF)
if(LGENGINE_BUILD_EXAMPLE)
  add_executable(LGEngineExample example.cpp)
  target_link_libraries(LGEngineExample PRIVATE LGEngine::Client gdi32 shell32)
  if(MSVC)
    target_compile_options(LGEngineExample PRIVATE /utf-8 /W4)
  else()
    target_link_options(LGEngineExample PRIVATE -municode -static)
    target_compile_options(LGEngineExample PRIVATE -Wall -Wextra)
  endif()
endif()
option(LGENGINE_BUILD_TESTS "Build the host-less wire test (golden bytes; needs no host)" OFF)
if(LGENGINE_BUILD_TESTS)
  add_executable(LGEngineWireTest "${CMAKE_CURRENT_SOURCE_DIR}/../../../tests/lgengine_wire_test.cpp")
  target_link_libraries(LGEngineWireTest PRIVATE LGEngine::Client)
  if(MSVC)
    target_compile_options(LGEngineWireTest PRIVATE /utf-8 /W4)
  else()
    target_link_options(LGEngineWireTest PRIVATE -static)
    target_compile_options(LGEngineWireTest PRIVATE -Wall -Wextra)
  endif()
endif()
