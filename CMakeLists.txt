cmake_minimum_required(VERSION 3.20)
project(LiquidGlassControl LANGUAGES C CXX RC)
add_subdirectory(integrations/LGEngine/client)
add_executable(LiquidGlassLGNative WIN32 integrations/LGEngine/native/main.cpp)
target_compile_features(LiquidGlassLGNative PRIVATE cxx_std_17)
target_link_libraries(LiquidGlassLGNative PRIVATE LGEngineCppClient gdi32 shell32)
set_target_properties(LiquidGlassLGNative PROPERTIES OUTPUT_NAME "LiquidGlass.LGPreview" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>/lgengine")
if(MSVC)
  target_compile_options(LiquidGlassLGNative PRIVATE /utf-8 /W4)
  set_property(TARGET LiquidGlassLGNative PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_link_options(LiquidGlassLGNative PRIVATE -municode -static)
endif()
add_library(MinHook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
add_library(GlassEngine SHARED src/glass_engine.cpp src/glass_engine.def)
target_compile_features(GlassEngine PRIVATE cxx_std_20)
set(LIQUIDGLASS_WINRT_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/build/winrt/include" CACHE PATH "Generated C++/WinRT and WinUI headers (prepare-winrt.ps1)")
if(NOT EXISTS "${LIQUIDGLASS_WINRT_INCLUDE}/winrt/Microsoft.UI.Xaml.Media.h" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/build/winrt/headers.zip")
  file(ARCHIVE_EXTRACT INPUT "${CMAKE_CURRENT_SOURCE_DIR}/build/winrt/headers.zip" DESTINATION "${LIQUIDGLASS_WINRT_INCLUDE}")
endif()
if(EXISTS "${LIQUIDGLASS_WINRT_INCLUDE}/winrt/Microsoft.UI.Xaml.Media.h")
  target_include_directories(GlassEngine BEFORE PRIVATE "${LIQUIDGLASS_WINRT_INCLUDE}")
elseif(MSVC)
  message(FATAL_ERROR "Run pwsh ./prepare-winrt.ps1 before configuring.")
endif()
target_link_libraries(GlassEngine PRIVATE MinHook ole32 oleaut32 uuid runtimeobject comctl32 dwmapi advapi32 d2d1 dxguid)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/engine-$<CONFIG>.rc" CONTENT "201 RCDATA \"$<TARGET_FILE:GlassEngine>\"\n202 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/release/Third Party Notices.txt\"\n")
add_executable(LiquidGlassControl WIN32 src/main.cpp assets/app.rc)
add_executable(LiquidGlassTests WIN32 src/main.cpp assets/app.rc)
target_compile_definitions(LiquidGlassTests PRIVATE LG_ENABLE_SELF_TEST)
target_sources(LiquidGlassTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/engine-$<CONFIG>.rc")
add_dependencies(LiquidGlassTests GlassEngine)
target_include_directories(LiquidGlassTests PRIVATE assets)
target_compile_features(LiquidGlassTests PRIVATE cxx_std_17)
target_link_libraries(LiquidGlassTests PRIVATE dwmapi shell32 user32 gdi32 advapi32 ole32 comctl32 uxtheme uuid)
target_sources(LiquidGlassControl PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/engine-$<CONFIG>.rc")
add_dependencies(LiquidGlassControl GlassEngine)
target_include_directories(LiquidGlassControl PRIVATE assets)
target_compile_features(LiquidGlassControl PRIVATE cxx_std_17)
target_link_libraries(LiquidGlassControl PRIVATE dwmapi shell32 user32 gdi32 advapi32 ole32 comctl32 uxtheme uuid)
if(MSVC)
  set_property(TARGET LiquidGlassTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(LiquidGlassTests PRIVATE /utf-8 /W4)
  set_property(TARGET LiquidGlassControl PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET MinHook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set_property(TARGET GlassEngine PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(GlassEngine PRIVATE /utf-8 /W4)
  target_compile_options(LiquidGlassControl PRIVATE /utf-8 /W4)
  foreach(target GlassEngine LiquidGlassControl LiquidGlassTests)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/Gw>)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> $<$<CONFIG:Release>:/INCREMENTAL:NO>)
  endforeach()
else()
  target_link_options(LiquidGlassTests PRIVATE -municode -static)
  target_compile_options(LiquidGlassTests PRIVATE -Wall -Wextra)
  target_link_options(LiquidGlassControl PRIVATE -municode -static)
  target_compile_options(LiquidGlassControl PRIVATE -Wall -Wextra)
  target_link_options(GlassEngine PRIVATE -static)
  target_compile_options(GlassEngine PRIVATE -Wall -Wextra)
  foreach(target GlassEngine LiquidGlassControl LiquidGlassTests)
    target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:-Os> $<$<CONFIG:Release>:-ffunction-sections> $<$<CONFIG:Release>:-fdata-sections>)
    target_link_options(${target} PRIVATE $<$<CONFIG:Release>:-Wl,--gc-sections,--strip-all>)
  endforeach()
endif()
